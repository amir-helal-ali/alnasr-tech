    55→
    56→type InvoiceStatus = 'draft' | 'issued' | 'submitted' | 'accepted' | 'paid' | 'cancelled';
    57→
    58→interface LineItem {
    59→  id: string;
    60→  description: string;
    61→  quantity: number;
    62→  unitPrice: number;
    63→  taxRate: number;
    64→}
    65→
    66→interface Invoice {
    67→  id: string;
    68→  invoiceNumber: string;
    69→  customerId: string;
    70→  customerName: string;
    71→  status: InvoiceStatus;
    72→  issueDate: string;
    73→  dueDate: string;
    74→  lineItems: LineItem[];
    75→  notes: string;
    76→  createdAt: string;
    77→}
    78→
    79→// ─── Helpers ─────────────────────────────────────────────────────────────────
    80→
    81→const EGYPT_VAT = 14;
    82→
    83→function calcLineTaxAmount(item: LineItem): number {
    84→  return item.quantity * item.unitPrice * (item.taxRate / 100);
    85→}
    86→
    87→function calcLineTotal(item: LineItem): number {
    88→  return item.quantity * item.unitPrice + calcLineTaxAmount(item);
    89→}
    90→
    91→function calcSubtotal(items: LineItem[]): number {
    92→  return items.reduce((sum, i) => sum + i.quantity * i.unitPrice, 0);
    93→}
    94→
    95→function calcTaxTotal(items: LineItem[]): number {
    96→  return items.reduce((sum, i) => sum + calcLineTaxAmount(i), 0);
    97→}
    98→
    99→function calcGrandTotal(items: LineItem[]): number {
   100→  return calcSubtotal(items) + calcTaxTotal(items);
   101→}
   102→
   103→function formatCurrency(amount: number): string {
   104→  return amount.toLocaleString('ar-EG', {
   105→    minimumFractionDigits: 2,
   106→    maximumFractionDigits: 2,
   107→  }) + ' ج.م';
   108→}
   109→
   110→function formatDate(dateStr: string): string {
   111→  return new Date(dateStr).toLocaleDateString('ar-EG', {
   112→    year: 'numeric',
   113→    month: 'short',
   114→    day: 'numeric',
   115→  });
   116→}
   117→
   118→function uid(): string {
   119→  return Math.random().toString(36).substring(2, 10);
   120→}

   329→    setFormNotes('');

   331→      { id: uid(), description: '', quantity: 1, unitPrice: 0, taxRate: EGYPT_VAT },
   332→    ]);
   333→    setEditingInvoice(null);
   334→  }, []);
   335→
   336→  // ─── Open create ────────────────────────────────────────────────────────
   337→
   338→  const openCreate = useCallback(() => {
   339→    resetForm();
   340→    setFormOpen(true);
   341→  }, [resetForm]);
   342→
   343→  // ─── Open edit ──────────────────────────────────────────────────────────
   344→

   346→    setEditingInvoice(invoice);

   348→    setFormIssueDate(invoice.issueDate);
   349→    setFormDueDate(invoice.dueDate);

   351→    setFormLineItems(
   352→      invoice.lineItems.map((li) => ({ ...li, id: uid() }))

   354→    setFormOpen(true);
   355→  }, []);
   356→
   357→  // ─── Open detail ────────────────────────────────────────────────────────
   358→
   359→  const openDetail = useCallback((invoice: Invoice) => {
   360→    setViewingInvoice(invoice);
   361→    setDetailOpen(true);
   362→  }, []);
   363→
   364→  // ─── Open delete ────────────────────────────────────────────────────────
   365→
   366→  const openDelete = useCallback((invoice: Invoice) => {
   367→    setDeletingInvoice(invoice);
   368→    setDeleteOpen(true);
   369→  }, []);

   372→
   373→  const addLineItem = useCallback(() => {
   374→    setFormLineItems((prev) => [
   375→      ...prev,

   377→    ]);
   378→  }, []);
   379→
   380→  const removeLineItem = useCallback((id: string) => {
   381→    setFormLineItems((prev) => {
   382→      if (prev.length <= 1) return prev;
   383→      return prev.filter((li) => li.id !== id);
   384→    });
   385→  }, []);
   386→
   387→  const updateLineItem = useCallback(
   388→    (id: string, field: keyof LineItem, value: string | number) => {
   389→      setFormLineItems((prev) =>
   390→        prev.map((li) => (li.id === id ? { ...li, [field]: value } : li))
   391→      );
   392→    },
   393→    []
   394→  );
   395→

   397→

   399→    // Validate
   400→    if (!formCustomerId) {
   401→      toast.error(isAr ? 'يرجى اختيار العميل' : 'Please select a customer');

   405→      toast.error(isAr ? 'يرجى إدخال تاريخ الإصدار' : 'Please enter issue date');

   464→    resetForm();

   473→    setDeletingInvoice(null);

   484→
   485→  // ─── Render ─────────────────────────────────────────────────────────────

   487→  return (

   489→      <div className="space-y-6">
   490→        {/* ── Header ──────────────────────────────────────────────────── */}
   491→        <div className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-4">
   492→          <div className="flex items-center gap-3">

   494→              <FileText className="h-5 w-5 text-primary" />
   495→            </div>
   496→            <div>
   497→              <h1 className="text-2xl font-bold text-foreground">{t.invoices}</h1>
   498→              <p className="text-sm text-muted-foreground">
   499→                {isAr ? 'إدارة الفواتير والضرائب' : 'Invoice & Tax Management'}
   500→              </p>
   501→            </div>
   502→          </div>
   503→          <Button onClick={openCreate} className="gap-2">
   504→            <Plus className="h-4 w-4" />

   509→        {/* ── Status Filter Tabs ──────────────────────────────────────── */}
   510→        <Tabs value={activeTab} onValueChange={setActiveTab}>
   511→          <TabsList className="flex-wrap h-auto gap-1 bg-muted/50 p-1">
   512→            <TabsTrigger value="all" className="gap-1.5 text-xs sm:text-sm">
   513→              {isAr ? 'الكل' : 'All'}

   515→                {tabCounts.all || 0}
   516→              </span>

   528→                  </span>

   530→              )

   537→          <CardContent className="p-0">
   538→            <ScrollArea className="w-full">
   539→              <Table>
