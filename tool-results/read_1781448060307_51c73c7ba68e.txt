   577→                              setCurrentPage((p) => Math.max(1, p - 1))
   578→                            }
   579→                            className={
   580→                              currentPage === 1

   583→                            }

   599→                            onClick={() =>
   600→                              setCurrentPage((p) => Math.min(totalPages, p + 1))

   602→                            className={

   605→                                : 'cursor-pointer'
   606→                            }
   607→                          />
   608→                        </PaginationItem>
   609→                      </PaginationContent>
   610→                    </Pagination>
   611→                  </div>
   612→                )}
   613→              </>
   614→            )}
   615→          </CardContent>
